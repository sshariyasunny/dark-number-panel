        function saveSentOtpKey(uniqueKey) {
            sentTelegramOtpKeys.add(uniqueKey);
            // লোকাল স্টোরেজে সংরক্ষণ (সর্বোচ্চ ২০০ মেসেজ কি সংরক্ষণ করবে মেমোরি ক্লিয়ার রাখতে)

        // 🚀 Telegram Notification Forwarder
        async function sendOtpToTelegram(item, otpCode, appDetails, country) {
            if (!APP_CONFIG.TELEGRAM_ENABLED) return;
            if (!APP_CONFIG.BOT_TOKEN || APP_CONFIG.BOT_TOKEN === "YOUR_TELEGRAM_BOT_TOKEN_HERE") return;
            if (!APP_CONFIG.GROUP_CHAT_ID || APP_CONFIG.GROUP_CHAT_ID === "YOUR_TELEGRAM_GROUP_CHAT_ID_HERE") return;

            // ইউনিক কী তৈরি
            const uniqueKey = `${item.number}_${otpCode}_${item.time || item.message}`;
            if (sentTelegramOtpKeys.has(uniqueKey)) {
                return; // অলরেডি সেন্ড হয়ে গেছে
            }

            const rawSms = item.message || "N/A";
            const number = item.number || "Unknown";
            
            let timeStr = "Just Now";
            if (item.time) {
                const dateObj = new Date(item.time);
                if (!isNaN(dateObj.getTime())) {
                    timeStr = dateObj.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit', second: '2-digit' });
                }
            }

            // সুন্দর ও প্রফেশনাল টেলিগ্রাম মেসেজ টেমপ্লেট
            const telegramMessage = 
`⚡ <b>STOCK OTP ZONE - NEW OTP RECEIVED</b> ⚡
━━━━━━━━━━━━━━━━━━━━━━
📱 <b>Phone Number:</b> <code>${number}</code>
🌍 <b>Country:</b> <b>${country}</b>
🏷️ <b>Service:</b> <b>${appDetails.name}</b>
🔑 <b>OTP Code:</b> <code>${otpCode}</code>
⏰ <b>Time:</b> ${timeStr}

📩 <b>Message:</b>
<i>${escapeHTML(rawSms)}</i>
━━━━━━━━━━━━━━━━━━━━━━
<i>🛡️ Automated Secure OTP System</i>`;

            const payload = {
                chat_id: APP_CONFIG.GROUP_CHAT_ID,
                text: telegramMessage,
                parse_mode: "HTML",
                reply_markup: {
                    inline_keyboard: [
                        [
                            {
                                text: APP_CONFIG.BUTTON_TEXT,
                                url: APP_CONFIG.MINI_BOT_URL
                            }
                        ]
                    ]
                }
            };

            try {
                const res = await fetch(`https://api.telegram.org/bot${APP_CONFIG.BOT_TOKEN}/sendMessage`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(payload)
                });
                
                if (res.ok) {
                    saveSentOtpKey(uniqueKey);
                } else {
                    const errRes = await res.json();
                    console.error("Telegram Forwarding Failed:", errRes);
                }
            } catch (err) {
                console.error("Telegram Request Error:", err);
            }
        }

        // এইচটিএমএল ক্যারেক্টার এস্কেপ ফাংশন
        function escapeHTML(str) {
            return str
                .replace(/&/g, "&amp;")
                .replace(/</g, "&lt;")
                .replace(/>/g, "&gt;");
        }

        // ভুলত্রুটি দেখানোর ফাংশন

        // Fetch Live OTPs and Check Active Sessions
        async function fetchLatestSMS() {
            if (!APP_CONFIG.API_TOKEN || APP_CONFIG.API_TOKEN === "") {
                showError("API Token সেট করা নেই। কোড চেক করুন।");
                return;
            }

            const banner = document.getElementById('error-banner');
            if (banner) banner.classList.add('hidden');
            const loadingEl = document.getElementById('loading-state');
            if (loadingEl) loadingEl.classList.add('hidden');

            const sessions = getValidSessions();
            const pending = sessions.filter(s => s.request_id && s.otp === "------");

            if (pending.length > 0) {
                const reqIds = pending.map(s => Number(s.request_id)).filter(id => !isNaN(id) && id > 0);
                if (reqIds.length > 0) {
                    try {
                        const response = await fetch(`${APP_CONFIG.BASE_URL}/check`, {
                            method: 'POST',
                            headers: {
                                'Content-Type': 'application/json',
                                'mknetwork-key': APP_CONFIG.API_TOKEN
                            },
                            body: JSON.stringify({ request_ids: reqIds })
                        });

                        if (response.ok) {
                            const result = await response.json();
                            if (result.data && result.data.results) {
                                let updated = false;
                                result.data.results.forEach(res => {
                                    if (res.status === 'success' && res.otp_code) {
                                        const s = sessions.find(item => item.request_id == res.request_id);
                                        if (s && s.otp !== res.otp_code) {
                                            s.otp = res.otp_code;
                                            s.sms = res.full_sms || `Your verification code is ${res.otp_code}`;
                                            updated = true;

                                            // Telegram Notification
                                            const appDetails = getAppDetails(s.sms);
                                            const country = s.country || getCountryByPrefix(s.full_number);
                                            sendOtpToTelegram({
                                                number: s.full_number,
                                                message: s.sms,
                                                time: Date.now()
                                            }, s.otp, appDetails, country);
                                        }
                                    }
                                });

                                if (updated) {
                                    localStorage.setItem('recent_otp_sessions', JSON.stringify(sessions));
                                    renderModalSessions();
                                }
                            }
                        }
                    } catch (e) {
                        console.warn("Check status error:", e);
                    }
                }
            }

            // Convert sessions with OTPs to feed cards
            const feedItems = sessions.filter(s => s.otp && s.otp !== "------").map(s => ({
                number: s.full_number,
                message: s.sms || `Verification code: ${s.otp}`,
                time: s.timestamp
            }));

            renderSMSCards(feedItems);
        }

        // ট্রাফিক ফিল্টারিং
        function getFilteredAndSortedTraffic(activeRanges) {
            let grouped = {};
            
            activeRanges.forEach(([range, hits]) => {
                const cleanRange = range.replace(/X/g, "");
                const country = getCountryByPrefix(cleanRange);
                if (!grouped[country]) grouped[country] = [];
                grouped[country].push({ range, hits });
            });

            let finalSelected = [];
            for (const country in grouped) {
                const topRanges = grouped[country]
                    .sort((a, b) => b.hits - a.hits)
                    .slice(0, 2);
                finalSelected.push(...topRanges);
            }

            return finalSelected
                .sort((a, b) => b.hits - a.hits)
                .slice(0, 8);
        }

        // Allocate Number API
        async function allocateNumberFromModal(rid, btnElement) {
            if (!APP_CONFIG.API_TOKEN) {
                alert("API Token কনফিগার করা নেই।");
                return;
            }
            
            const originalText = btnElement.innerHTML;
            btnElement.disabled = true;
            btnElement.innerHTML = `<i class="fa-solid fa-spinner animate-spin"></i>`;

            let attempts = 0;
            let allocatedData = null;
            const targetRange = rid.includes('X') ? rid : (rid + 'XXX');

            while (attempts < 3) {
                attempts++;
                try {
                    const response = await fetch(`${APP_CONFIG.BASE_URL}/getnum`, {
                        method: 'POST',
                        headers: {
                            'Content-Type': 'application/json',
                            'mknetwork-key': APP_CONFIG.API_TOKEN
                        },
                        body: JSON.stringify({ range: targetRange })
                    });

                    if (response.ok) {
                        const result = await response.json();
                        if (result.meta && result.meta.code === 200 && result.data) {
                            allocatedData = result.data;
                            break;
                        }
                    } else if (response.status === 404 || response.status === 422) {
                        break;
                    }
                } catch (error) {
                    console.error("Allocation attempt failed:", error);
                }
            }

            if (allocatedData) {
                const fullNum = allocatedData.full_number || allocatedData.number;
                const sessionData = {
                    request_id: allocatedData.request_id || allocatedData.id || Date.now(),
                    full_number: fullNum,
                    no_plus_number: fullNum ? fullNum.replace(/\D/g, "") : "",
                    country: allocatedData.country || getCountryByPrefix(fullNum || ""),
                    operator: allocatedData.operator || "Auto Route",
                    timestamp: Date.now(),
                    app: currentActiveApp,
                    otp: allocatedData.otp_code || "------",
                    sms: allocatedData.full_sms || ""
                };
                
                addSessionToHistory(sessionData);
                renderModalSessions();
                fetchLatestSMS();
            } else {
                alert("দুঃখিত, এই রেঞ্জে এই মুহূর্তে কোনো নম্বর খালি নেই। অনুগ্রহ করে অন্য দেশ বা রেঞ্জ নির্বাচন করুন।");
            }

            btnElement.disabled = false;
            btnElement.innerHTML = originalText;
        }

        // ওটিপি ম্যাচিং
        function checkActiveSessionOTP(smsList) {
            let sessions = getValidSessions();
            if (sessions.length === 0) {
                const container = document.getElementById('modal-active-session-container');
                if (container) container.classList.add('hidden');
                return;
            }

            let updated = false;

            sessions.forEach(sessionData => {
                const cleanSessionNum = sessionData.full_number.replace(/\D/g, "");
                const matchedItem = smsList.find(item => {
                    const cleanItemNum = (item.number || "").replace(/\D/g, "");
                    return cleanItemNum.includes(cleanSessionNum) || cleanSessionNum.includes(cleanItemNum);
                });

                if (matchedItem) {
                    const otp = extractOTP(matchedItem.message || "");
                    if (otp !== "------" && sessionData.otp !== otp) {
                        sessionData.otp = otp;
                        sessionData.sms = matchedItem.message || "";
                        updated = true;
                    }
                }
            });

            if (updated || smsList.length > 0) {
                localStorage.setItem('recent_otp_sessions', JSON.stringify(sessions));
            }

            renderModalSessions();
        }

        function getCountryByPrefix(number) {
            if (!number) return "Global Route 🌐";
            const clean = number.replace(/\D/g, '');
            const sortedPrefixes = Object.keys(countryCodes).sort((a, b) => b.length - a.length);
            
            for (const prefix of sortedPrefixes) {
                if (clean.startsWith(prefix)) {
                    return countryCodes[prefix];
                }
            }
            return "Global Route 🌐";
        }

        function extractOTP(sms) {
            if (!sms) return "------";
            
            let matches = sms.match(/\b\d{4,8}\b/);
            if (matches) return matches[0];

            let spacedMatches = sms.match(/\b\d{3}[\s-]\d{3}\b/);
            if (spacedMatches) {
                return spacedMatches[0].replace(/[\s-]/g, "");
            }

            const codePatterns = [
                /(?:code|otp|verification|g-)\s*(?:is|:)?\s*(\d{4,8})/i,
                /(\d{3}[\s-]\d{3})/
            ];
            for (let pattern of codePatterns) {
                let pMatch = sms.match(pattern);
                if (pMatch) {
                    return pMatch[1].replace(/[\s-]/g, "");
                }
            }

            let fallbackMatches = sms.match(/\d{4,8}/);
            if (fallbackMatches) return fallbackMatches[0];

            return "------";
        }

        function copyToClipboard(text) {
            if (navigator.clipboard && navigator.clipboard.writeText) {
                navigator.clipboard.writeText(text).then(() => {
                    showCustomToast(text);
                }).catch(err => {
                    fallbackCopy(text);
                });
            } else {
                fallbackCopy(text);
            }
        }

        function fallbackCopy(text) {
            const textArea = document.createElement("textarea");
            textArea.value = text;
            textArea.style.position = "fixed";
            document.body.appendChild(textArea);
            textArea.focus();
            textArea.select();
            try {
                document.execCommand('copy');
                showCustomToast(text);
            } catch (err) {
                console.error('Copy failed: ', err);
            }
            document.body.removeChild(textArea);
        }

        function showCustomToast(code) {
            const toast = document.getElementById('copy-toast');
            const toastText = document.getElementById('toast-text');
            if (toast && toastText) {
                toastText.innerText = `Code Copied: ${code}`;
                toast.classList.remove('opacity-0', 'pointer-events-none', 'scale-90');
                toast.classList.add('opacity-100', 'scale-100');
                
                setTimeout(() => {
                    toast.classList.remove('opacity-100', 'scale-100');
                    toast.classList.add('opacity-0', 'pointer-events-none', 'scale-90');
                }, 2500);
            }
        }

        function filterCodes() {
            const query = document.getElementById('search-input').value.toLowerCase();
            const cards = document.getElementsByClassName('sms-card');
            
            Array.from(cards).forEach(card => {
                const text = card.textContent.toLowerCase();
                if (text.includes(query)) {
                    card.style.display = "block";
                } else {
                    card.style.display = "none";
                }
            });
        }
