        function getCountryInfo(identifier) {
            if (!identifier) return { name: "Global Route", flag: "🌐", fullText: "Global Route 🌐" };
            
            const clean = identifier.toString().replace(/\D/g, '');
            const sortedPrefixes = Object.keys(countryCodes).sort((a, b) => b.length - a.length);

            for (const prefix of sortedPrefixes) {
                if (clean.startsWith(prefix)) {
                    const c = countryCodes[prefix];
                    return {
                        name: c.name,
                        flag: c.flag,
                        fullText: `${c.flag} ${c.name}`
                    };
                }
            }

            const lower = identifier.toString().toLowerCase();
            for (const prefix of Object.keys(countryCodes)) {
                const c = countryCodes[prefix];
                if (lower.includes(c.name.toLowerCase())) {
                    return {
                        name: c.name,
                        flag: c.flag,
                        fullText: `${c.flag} ${c.name}`
                    };
                }
            }

            return { name: identifier, flag: "🌐", fullText: `🌐 ${identifier}` };
        }

        // 1. Fetch Latest SMS & Verify Active Sessions
        async function fetchLatestSMS() {
            if (!APP_CONFIG.API_TOKEN) return;

            const sessions = getValidSessions();
            const pending = sessions.filter(s => s.request_id && (!s.otp || s.otp === "------"));

            if (pending.length > 0) {
                const reqIds = pending.map(s => Number(s.request_id)).filter(id => !isNaN(id) && id > 0);
                if (reqIds.length > 0) {
                    try {
                        const response = await fetch(`${APP_CONFIG.BASE_URL}/check/status`, {
                            method: 'POST',
                            headers: {
                                'Content-Type': 'application/json',
                                'mknetwork-key': APP_CONFIG.API_TOKEN
                            },
                            body: JSON.stringify({ request_ids: reqIds })
                        });

                        if (response.ok) {
                            const result = await response.json();
                            if (result.data && result.data.results) {
                                let updated = false;
                                result.data.results.forEach(res => {
                                    if (res.status === 'success' && res.otp_code) {
                                        const s = sessions.find(item => item.request_id == res.request_id);
                                        if (s && s.otp !== res.otp_code) {
                                            s.otp = res.otp_code;
                                            s.sms = res.full_sms || `Your verification code is ${res.otp_code}`;
                                            updated = true;

                                            const appDetails = getAppDetails(s.sms);
                                            const cInfo = getCountryInfo(s.full_number || s.country);
                                            sendOtpToTelegram({
                                                number: s.full_number,
                                                message: s.sms,
                                                time: Date.now()
                                            }, s.otp, appDetails, cInfo.fullText);

                                            playOtpArrivalSound();
                                        }
                                    }
                                });

                                if (updated) {
                                    localStorage.setItem('recent_otp_sessions', JSON.stringify(sessions));
                                    renderModalSessions();
                                }
                            }
                        }
                    } catch (e) {
                        console.warn("Check status error:", e);
                    }
                }
            }

            const feedItems = sessions.filter(s => s.otp && s.otp !== "------").map(s => ({
                number: s.full_number,
                message: s.sms || `Verification code: ${s.otp}`,
                time: s.timestamp
            }));

            renderSMSCards(feedItems);
        }

                const range = it.phone_range || it.range;
                const country = it.country;
                const operator = it.operator || 'Auto Route';
                const health = it.health || 'good';

        // Session Memory

        // 4. Number Allocation Core via /getnum/number
        async function allocateNumberFromModal(range, btnElement) {
            if (!APP_CONFIG.API_TOKEN) {
                alert("API Token is missing. Please check configuration.");
                return;
            }
            
            const originalHTML = btnElement.innerHTML;
            btnElement.disabled = true;
            btnElement.innerHTML = `<i class="fa-solid fa-circle-notch animate-spin text-xs"></i> <span>ALLOCATING...</span>`;

            let allocatedData = null;

            try {
                const response = await fetch(`${APP_CONFIG.BASE_URL}/getnum/number`, {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                        'mknetwork-key': APP_CONFIG.API_TOKEN
                    },
                    body: JSON.stringify({ range: range })
                });

                if (response.ok) {
                    const result = await response.json();
                    if (result.meta && result.meta.code === 200 && result.data) {
                        allocatedData = result.data;
                    }
                } else {
                    const err = await response.json();
                    alert(err.message || "Failed to allocate number. Please try another range.");
                }
            } catch (error) {
                console.error("Allocation error:", error);
                alert("Network error while allocating number. Please retry.");
            }

            if (allocatedData) {
                const fullNum = allocatedData.full_number || allocatedData.number;
                const cInfo = getCountryInfo(allocatedData.country || fullNum || range);

                const sessionData = {
                    request_id: allocatedData.request_id,
                    full_number: fullNum,
                    no_plus_number: fullNum ? fullNum.replace(/\D/g, "") : "",
                    country: cInfo.fullText,
                    operator: allocatedData.operator || "Auto Route",
                    timestamp: Date.now(),
                    app: currentActiveApp,
                    otp: allocatedData.otp_code || "------",
                    sms: allocatedData.full_sms || ""
                };
                
                addSessionToHistory(sessionData);
                renderModalSessions();
                fetchLatestSMS();
                
                playAllocateSound();

                showCustomToast(`Allocated: ${fullNum}`);
            }

            btnElement.disabled = false;
            btnElement.innerHTML = originalHTML;
        }

        function getFormattedNumber(num) {
            if (!num) return "Unknown";
            const clean = num.replace(/\D/g, '');
            if (clean.length < 6) return num;
            const prefix = clean.slice(0, 3);
            const suffix = clean.slice(-3);
            return `+${prefix}••••••${suffix}`;
        }

        function getAppDetails(sms) {
            const lowerSms = (sms || "").toLowerCase();
            if (lowerSms.includes('facebook') || lowerSms.includes('fb')) {
                return { name: 'FACEBOOK', icon: '<i class="fa-brands fa-facebook text-[#1877F2] text-sm animate-pulse"></i>', colorClass: 'text-[#1877F2]' };
            }
            if (lowerSms.includes('instagram') || lowerSms.includes('ig') || lowerSms.includes('insta')) {
                return { name: 'INSTAGRAM', icon: '<i class="fa-brands fa-instagram text-[#E1306C] text-sm animate-pulse"></i>', colorClass: 'text-[#E1306C]' };
            }
            if (lowerSms.includes('whatsapp') || lowerSms.includes('wa')) {
                return { name: 'WHATSAPP', icon: '<i class="fa-brands fa-whatsapp text-[#25D366] text-sm animate-pulse"></i>', colorClass: 'text-[#25D366]' };
            }
            if (lowerSms.includes('telegram') || lowerSms.includes('tg')) {
                return { name: 'TELEGRAM', icon: '<i class="fa-brands fa-telegram text-[#0088cc] text-sm animate-pulse"></i>', colorClass: 'text-[#0088cc]' };
            }
            if (lowerSms.includes('twitter') || lowerSms.includes(' x ')) {
                return { name: 'X APP', icon: '<i class="fa-brands fa-x-twitter text-white text-sm animate-pulse"></i>', colorClass: 'text-white' };
            }
            if (lowerSms.includes('google') || lowerSms.includes('g-')) {
                return { name: 'GOOGLE', icon: '<i class="fa-brands fa-google text-[#4285F4] text-sm animate-pulse"></i>', colorClass: 'text-[#4285F4]' };
            }
            if (lowerSms.includes('tiktok')) {
                return { name: 'TIKTOK', icon: '<i class="fa-brands fa-tiktok text-[#EE1D52] text-sm animate-pulse"></i>', colorClass: 'text-[#EE1D52]' };
            }
            return { name: 'SECURE OTP', icon: '<i class="fa-solid fa-shield-halved text-cyan-400 text-xs animate-pulse"></i>', colorClass: 'text-cyan-400' };
        }

        function extractOTP(sms) {
            if (!sms) return "------";
            let matches = sms.match(/\b\d{4,8}\b/);
            if (matches) return matches[0];

            let spacedMatches = sms.match(/\b\d{3}[\s-]\d{3}\b/);
            if (spacedMatches) return spacedMatches[0].replace(/[\s-]/g, "");

            const codePatterns = [
                /(?:code|otp|verification|g-)\s*(?:is|:)?\s*(\d{4,8})/i,
                /(\d{3}[\s-]\d{3})/
            ];
            for (let pattern of codePatterns) {
                let pMatch = sms.match(pattern);
                if (pMatch) return pMatch[1].replace(/[\s-]/g, "");
            }

            let fallbackMatches = sms.match(/\d{4,8}/);
            if (fallbackMatches) return fallbackMatches[0];

            return "------";
        }

        function copyToClipboard(text) {
            if (!text || text === "------") return;

            if (navigator.clipboard && navigator.clipboard.writeText) {
                navigator.clipboard.writeText(text).then(() => {
                    showCustomToast(text);
                }).catch(() => fallbackCopy(text));
            } else {
                fallbackCopy(text);
            }
        }

        function fallbackCopy(text) {
            const textArea = document.createElement("textarea");
            textArea.value = text;
            textArea.style.position = "fixed";
            document.body.appendChild(textArea);
            textArea.focus();
            textArea.select();
            try {
                document.execCommand('copy');
                showCustomToast(text);
            } catch (err) {
                console.error('Copy fallback failed: ', err);
            }
            document.body.removeChild(textArea);
        }

        function showCustomToast(code) {
            const toast = document.getElementById('copy-toast');
            const toastText = document.getElementById('toast-text');
            if (toast && toastText) {
                toastText.innerText = `COPIED: ${code}`;
                toast.classList.remove('opacity-0', 'pointer-events-none', 'scale-90');
                toast.classList.add('opacity-100', 'scale-100');
                
                setTimeout(() => {
                    toast.classList.remove('opacity-100', 'scale-100');
                    toast.classList.add('opacity-0', 'pointer-events-none', 'scale-90');
                }, 2200);
            }
        }

        function filterCodes() {
            const query = (document.getElementById('search-input')?.value || "").toLowerCase();
            const cards = document.getElementsByClassName('sms-card');
            
            Array.from(cards).forEach(card => {
                const text = card.textContent.toLowerCase();
                if (text.includes(query)) {
                    card.style.display = "flex";
                } else {
                    card.style.display = "none";
                }
            });
        }
